cmake_minimum_required(VERSION 3.24)
project(bitovp_vmaf LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BITOVP_CUDA "Build the owned CUDA extractor" OFF)
option(BITOVP_VULKAN "Build the owned Vulkan extractor" OFF)
option(BITOVP_TESTS "Build native correctness tests" ON)
option(BITOVP_CUDA_BASELINE_TESTS "Build slow device-scalar baselines for optimization comparisons" OFF)
set(LIBVMAF_SOURCE_DIR "" CACHE PATH "Pinned official libvmaf source directory")
set(LIBVMAF_BUILD_DIR "" CACHE PATH "Pinned official libvmaf build directory")
set(VMAF_MODEL_DIR "" CACHE PATH "Official model directory used by tests")
set(VMAF_DECODE_FIXTURE_DIR "" CACHE PATH "Optional directory with real codec fixtures for resident decoder tests")

if(NOT EXISTS "${LIBVMAF_SOURCE_DIR}/src/feature/feature_extractor.h" OR
   NOT EXISTS "${LIBVMAF_BUILD_DIR}/src/config.h")
  message(FATAL_ERROR "Set LIBVMAF_SOURCE_DIR and LIBVMAF_BUILD_DIR to the pinned official CPU build")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
find_package(Threads REQUIRED)
if(NOT EXISTS "${LIBVMAF_BUILD_DIR}/src/libvmaf.a")
  message(FATAL_ERROR "The official CPU static library is missing")
endif()
file(READ "${LIBVMAF_BUILD_DIR}/src/config.h" VMAF_CONFIGURATION)
if(VMAF_CONFIGURATION MATCHES "#define HAVE_CUDA 1")
  message(FATAL_ERROR "The oracle must be the official CPU-only libvmaf build")
endif()
add_library(official_vmaf STATIC IMPORTED)
set_target_properties(official_vmaf PROPERTIES
  IMPORTED_LOCATION "${LIBVMAF_BUILD_DIR}/src/libvmaf.a"
  INTERFACE_INCLUDE_DIRECTORIES "${LIBVMAF_SOURCE_DIR}/include;${LIBVMAF_BUILD_DIR}/include")

add_library(bitovp_oracle STATIC src/model_oracle.cpp src/oracle/official_adapter.c)
target_include_directories(bitovp_oracle PUBLIC include src
  PRIVATE "${LIBVMAF_SOURCE_DIR}/src" "${LIBVMAF_BUILD_DIR}/src")
target_link_libraries(bitovp_oracle PUBLIC official_vmaf Threads::Threads ${CMAKE_DL_LIBS} m)
target_compile_options(bitovp_oracle PRIVATE -Wall -Wextra -Werror)

add_library(bitovp_decode STATIC src/decoder.cpp)
target_include_directories(bitovp_decode PUBLIC include src)
target_link_libraries(bitovp_decode PUBLIC PkgConfig::AV)
target_compile_options(bitovp_decode PRIVATE -Wall -Wextra -Werror)

add_executable(bitovp-vmaf src/main.cpp)
target_link_libraries(bitovp-vmaf PRIVATE bitovp_oracle bitovp_decode)
target_compile_options(bitovp-vmaf PRIVATE -Wall -Wextra -Werror)

# Bind receipts to the complete owned source, including headers and shaders.
file(GLOB_RECURSE OWNED_SOURCES CONFIGURE_DEPENDS
  LIST_DIRECTORIES false RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  include/*.hpp src/*.hpp src/*.h src/*.cpp src/*.c src/*.cu src/*.cuh
  src/*.comp src/*.glsl src/*.json src/*.py tests/*.cpp tests/*.cu tests/*.py)
list(APPEND OWNED_SOURCES build.py benchmark.py upstream.lock.json)
list(SORT OWNED_SOURCES)
set(SOURCE_HASH_INPUT "")
foreach(source IN LISTS OWNED_SOURCES)
  configure_file("${source}" "source-receipt/${source}" COPYONLY)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND SOURCE_HASH_INPUT "${source}:${source_hash}\n")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" cmake_hash)
string(APPEND SOURCE_HASH_INPUT "CMakeLists.txt:${cmake_hash}\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LIBVMAF_BUILD_DIR}/src/libvmaf.a")
file(SHA256 "${LIBVMAF_BUILD_DIR}/src/libvmaf.a" oracle_hash)
string(APPEND SOURCE_HASH_INPUT "official-cpu-library:${oracle_hash}\n")
string(SHA256 OWNED_SOURCE_SHA256 "${SOURCE_HASH_INPUT}")
target_compile_definitions(bitovp-vmaf PRIVATE
  BITOVP_NATIVE_SOURCE_SHA256="${OWNED_SOURCE_SHA256}")

if(BITOVP_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(bitovp_cuda STATIC src/cuda/cuda_backend.cu src/cuda/parallel_speed.cu
    src/cuda/parallel_cambi.cu src/cuda/frame_conformer.cu src/cuda/parallel_adm.cu)
  target_include_directories(bitovp_cuda PUBLIC include)
  target_link_libraries(bitovp_cuda PUBLIC CUDA::cudart)
  set_target_properties(bitovp_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(bitovp_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(bitovp-vmaf PRIVATE bitovp_cuda)
  target_link_libraries(bitovp_decode PUBLIC bitovp_cuda CUDA::cuda_driver)
  target_compile_definitions(bitovp_decode PRIVATE BITOVP_HAVE_CUDA=1)
  target_compile_definitions(bitovp-vmaf PRIVATE BITOVP_HAVE_CUDA=1)
endif()

if(BITOVP_VULKAN)
  find_package(Vulkan REQUIRED)
  find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
  set(VULKAN_SPV_HEADERS "")
  foreach(shader psnr cambi speed adm motion)
    set(header "${CMAKE_CURRENT_BINARY_DIR}/generated/vulkan_${shader}_spv.hpp")
    add_custom_command(OUTPUT "${header}"
      COMMAND "${GLSLANG_VALIDATOR}" -V "${CMAKE_CURRENT_SOURCE_DIR}/src/vulkan/${shader}.comp"
        --vn bitovp_vulkan_${shader}_spv -o "${header}"
      DEPENDS "src/vulkan/${shader}.comp" "src/vulkan/speed_matrix.glsl" VERBATIM)
    list(APPEND VULKAN_SPV_HEADERS "${header}")
  endforeach()
  add_library(bitovp_vulkan STATIC src/vulkan/backend.cpp ${VULKAN_SPV_HEADERS})
  target_include_directories(bitovp_vulkan PUBLIC include
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(bitovp_vulkan PUBLIC Vulkan::Vulkan)
  target_compile_options(bitovp_vulkan PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  target_link_libraries(bitovp-vmaf PRIVATE bitovp_vulkan)
  target_compile_definitions(bitovp-vmaf PRIVATE BITOVP_HAVE_VULKAN=1)
endif()

if(BITOVP_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME official_source_integrity COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/source_test.py")
  add_test(NAME benchmark_process_cleanup COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_test.py")
  add_executable(model_oracle_test tests/model_oracle_test.cpp)
  target_link_libraries(model_oracle_test PRIVATE bitovp_oracle)
  target_compile_options(model_oracle_test PRIVATE -Wall -Wextra -Werror)
  if(VMAF_MODEL_DIR)
    add_test(NAME official_model_oracle COMMAND model_oracle_test "${VMAF_MODEL_DIR}")
    add_test(NAME native_runner COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/runner_test.py"
      --binary "$<TARGET_FILE:bitovp-vmaf>" --models "${VMAF_MODEL_DIR}")
  endif()
  if(BITOVP_CUDA)
    add_executable(cuda_decoder_test tests/cuda_decoder_test.cpp)
    target_link_libraries(cuda_decoder_test PRIVATE bitovp_decode)
    target_compile_options(cuda_decoder_test PRIVATE -Wall -Wextra -Werror)
    if(VMAF_DECODE_FIXTURE_DIR)
      add_test(NAME cuda_resident_decoders COMMAND cuda_decoder_test "${VMAF_DECODE_FIXTURE_DIR}")
    endif()
    if(BITOVP_CUDA_BASELINE_TESTS)
      add_library(bitovp_cuda_baseline STATIC src/cuda/scalar_features.cu)
      target_include_directories(bitovp_cuda_baseline PUBLIC include src)
      target_link_libraries(bitovp_cuda_baseline PUBLIC CUDA::cudart)
      set_target_properties(bitovp_cuda_baseline PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
      target_compile_options(bitovp_cuda_baseline PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
      foreach(test adm speed)
        add_executable(cuda_parallel_${test}_test tests/cuda_parallel_${test}_test.cpp)
        target_include_directories(cuda_parallel_${test}_test PRIVATE src/cuda)
        target_link_libraries(cuda_parallel_${test}_test PRIVATE bitovp_cuda bitovp_cuda_baseline bitovp_oracle)
        if(VMAF_MODEL_DIR)
          add_test(NAME cuda_parallel_${test} COMMAND cuda_parallel_${test}_test "${VMAF_MODEL_DIR}")
        endif()
      endforeach()
    endif()
    add_executable(cuda_frame_conformer_test tests/cuda_frame_conformer_test.cu)
    target_include_directories(cuda_frame_conformer_test PRIVATE src)
    target_link_libraries(cuda_frame_conformer_test PRIVATE bitovp_cuda)
    set_target_properties(cuda_frame_conformer_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    add_test(NAME cuda_resident_frame COMMAND cuda_frame_conformer_test)
    foreach(test cuda_psnr cuda_motion)
      add_executable(${test}_test tests/${test}_test.cpp)
      target_link_libraries(${test}_test PRIVATE bitovp_cuda)
      target_compile_options(${test}_test PRIVATE -Wall -Wextra -Werror)
      add_test(NAME ${test} COMMAND ${test}_test)
    endforeach()
    add_executable(cuda_features_test tests/cuda_features_test.cpp)
    target_link_libraries(cuda_features_test PRIVATE bitovp_cuda bitovp_oracle)
    target_compile_options(cuda_features_test PRIVATE -Wall -Wextra -Werror)
    if(VMAF_MODEL_DIR)
      add_test(NAME cuda_features COMMAND cuda_features_test "${VMAF_MODEL_DIR}")
      add_executable(cuda_cambi_test tests/cuda_cambi_test.cpp)
      target_link_libraries(cuda_cambi_test PRIVATE bitovp_cuda bitovp_oracle)
      target_compile_options(cuda_cambi_test PRIVATE -Wall -Wextra -Werror)
      add_test(NAME cuda_parallel_cambi COMMAND cuda_cambi_test "${VMAF_MODEL_DIR}")
    endif()
  endif()
  if(BITOVP_VULKAN)
    add_executable(vulkan_backend_test tests/vulkan_backend_test.cpp)
    target_link_libraries(vulkan_backend_test PRIVATE bitovp_vulkan)
    target_compile_options(vulkan_backend_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME vulkan_backend COMMAND vulkan_backend_test)
    add_executable(vulkan_motion_test tests/vulkan_motion_test.cpp)
    target_link_libraries(vulkan_motion_test PRIVATE bitovp_vulkan)
    target_compile_options(vulkan_motion_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME vulkan_motion COMMAND vulkan_motion_test)
    add_executable(vulkan_features_test tests/vulkan_features_test.cpp)
    target_link_libraries(vulkan_features_test PRIVATE bitovp_vulkan bitovp_oracle)
    target_compile_options(vulkan_features_test PRIVATE -Wall -Wextra -Werror)
    if(VMAF_MODEL_DIR)
      add_test(NAME vulkan_features COMMAND vulkan_features_test "${VMAF_MODEL_DIR}")
      add_executable(vulkan_adm_test src/vulkan/adm_test.cpp)
      target_link_libraries(vulkan_adm_test PRIVATE bitovp_vulkan bitovp_oracle)
      target_compile_options(vulkan_adm_test PRIVATE -Wall -Wextra -Werror)
      add_test(NAME vulkan_adm COMMAND vulkan_adm_test "${VMAF_MODEL_DIR}")
    endif()
  endif()
endif()

install(TARGETS bitovp-vmaf RUNTIME DESTINATION bin)
